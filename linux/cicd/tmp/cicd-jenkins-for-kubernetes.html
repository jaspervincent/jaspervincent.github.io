deploy/Jenkinsfile

#!groovy
node('master') {
    def jobname = env.JOB_BASE_NAME
    def name = jobname.split(/(dev-|qa-|prod-)/)[1]
    def workdir = env.WORKSPACE
    def port = env.PORT
    def nacosserver = env.Nacos_Server
    def nacosns = env.Nacos_Namespace
    def dockerid = env.BUILD_TAG
    def appenv = env.SVC_ENV
    def k8sns = env.k8s_namespace
    def k8sreplicas = env.replicas
    def skywalking = env.skywalking

    stage('代码获取') {
        dir("${workdir}") {
            checkout scm
        }
    }

    stage('构建代码') {
        sh "mvn clean package"
        sh "cp ${name}/target/${name}.jar deploy"
    }

    stage('构建Docker镜像') {
        docker.withRegistry('https://registry-vpc.cn-beijing.aliyuncs.com', 'e87917f6-1219-4d80-8c3e-d054ecbe268d') {
            def Image = docker.build("registry-vpc.cn-beijing.aliyuncs.com/xxx-${appenv}/${name}:latest", "./deploy --build-arg JAR_FILE=${name}.jar --build-arg SERVER_NAME=${name} --build-arg PORT=${port} --build-arg ENV=${appenv} --build-arg NACOS_SERVER=${nacosserver} --build-arg NACOS_NS=${nacosns} --build-arg SKYWALKING_SERVER=${skywalking}")
            Image.push("${dockerid}")
            Image.push("latest")
        }
    }

    try {
        stage('卸载服务') {
            sh "helm --kubeconfig /root/.kube/cici/${appenv}-config uninstall ${name} -n ${k8sns}"
            sh "sed -i 's/service-name/${name}/g' deploy/helm/Chart.yaml"
        }
    } catch (Exception e) {
        echo "continue"
    }

    stage('发布服务') {
        sh "helm --kubeconfig /root/.kube/cici/${appenv}-config upgrade ${name} deploy/helm \
             --install \
             --timeout 5m0s \
             --namespace ${k8sns} \
             --set service.port=${port} \
             --set name=${name} \
             --set replicas=${k8sreplicas} \
             --set deployment.image.registry=registry-vpc.cn-beijing.aliyuncs.com/cici-${appenv}/${name} \
             --set deployment.image.version=${dockerid}"
    }

    stage('服务列表') {
        sh "helm --kubeconfig /root/.kube/cici/${appenv}-config list -A | grep ${name}"
    }
}

deploy/down_nacos.sh 

ip=`ifconfig  | grep inet | awk '{print $2}' | grep -v "127.0.0.1"`
serviceName=`ls /app | grep *.jar | awk -F '.' '{print $1}'`
port=`ps -ef | grep java | grep -v grep  | awk -F '--server.port=' '{print $2}' | awk '{print $1}'`
namespacesId=`ps -ef | grep java | grep -v grep  | awk -F '--spring.cloud.nacos.config.namespace=' '{print $2}' | awk '{print $1}'`
nacosServer=`ps -ef | grep java | grep -v grep  | awk -F '--spring.cloud.nacos.config.server-addr=' '{print $2}' | awk '{print $1}'`

curl -X PUT -d "serviceName=$serviceName" -d "ip=$ip" -d "port=$port" -d "enabled=false" -d "namespacesId=$namespacesId" $nacosServer/nacos/v1/ns/instance\?serviceName\=$serviceName\&ip\=$ip\&port\=$port\&enabled\=false\&namespaceId\=$namespacesId

sleep 30

{{- end}}

cat <<\EOF> /usr/bin/k8s_log_aws_uswest2_dev
job_name=$1
config="-n dev1 --kubeconfig /mnt/jenkins/config_aws_uswest2_dev"
pod_name=$(kubectl ${config} get pod --sort-by=.metadata.creationTimestamp -l app=${job_name}|tail -1|awk '{print $1}')
# 检测pod启动后再获取log
while true; do
kubectl ${config} get pod $pod_name 2>/dev/null |grep -q Running 
  if [ $? -eq 0 ];then
      echo $pod_name "is running"
      break 
  fi
  echo $pod_name "waiting to running"
  sleep 1
done
kubectl ${config} logs -f $pod_name &
#jenkins无需自己解决子进程问题
#pid=$!
while true; do
  restarts=$(kubectl $config get pod $pod_name 2>/dev/null|grep -v NAME|awk '{print $4}')
  if [ $restarts -gt 0 ];then
      #kill $pid
      exit 1
  fi
  kubectl $config rollout status deployment $job_name --watch=false 2>/dev/null|grep -q successfully
  if [ $? -eq 0 ];then
      #kill $pid
      exit 0
  fi
  sleep 1
done
EOF

def V_GIT_URL = 'git@gitlab.xx.com:business/gateway.git'

pipeline {
    agent any
    tools {
        jdk 'jdk 17'
        maven 'maven 3.9'
    }
    options {
        withBuildUser()
    }
    environment {
        BUSINESS = "rl"
        DEPLOY_ENV = "dev"
        
        jobname = "${JOB_BASE_NAME}"
        OPS_GIT_URL = 'git@gitlab.xx.com:gitops/ops-deploy.git'
        OPS_PATH = "ops-deploy/${BUSINESS}/${jobname}/overlays/${DEPLOY_ENV}"
        image_url="public.ecr.aws/xxx/${jobname}:v${BUILD_NUMBER}"

        ARGO_APP = "${jobname}" + "-${DEPLOY_ENV}"
        ARGOCD_SERVER   = 'argocd.xxx.com:80'
        GITOPS_SHA = 'HEAD'
    }


    stages {
       stage('删除工作目录') {
            steps {
                // deleteDir()
                echo 'no delete'
           }
        }
        
        stage('拉取代码') {
            when {
                not {
                    environment name: 'GIT_BRANCH', value: ''

                    currentBuild.displayName = "#${BUILD_NUMBER} ${GIT_BRANCH} by ${env.BUILD_USER}"

                    git branch: env.GIT_BRANCH, credentialsId: 'git-ci', url: V_GIT_URL
                }
            }
        }

        stage('mvn编译构建') {
            steps {
                script {
                    echo "开始打包api-doc应用"
                    sh "mvn clean install"
                }
            }
        }

        stage('登录AWS ECR Public') {
            steps {

回溯脚本

pipeline {
    agent any
    options {
        withBuildUser()
        timestamps()
    }
    parameters {
        string(name: 'BUSINESS', defaultValue: "rl", description: '项目名称，例如 rl', trim: true)
        string(name: 'DEPLOY_ENV', defaultValue: "dev", description: '环境，例如 dev, prod', trim: true)
        string(name: 'SERVICE_NAME', defaultValue: "pc-web", description: '服务名称，例如 pc-web', trim: true)
        string(name: 'ROLLBACK_TAG', defaultValue: "v32", description: '要回滚的目标镜像TAG，例如 v32', trim: true)
        booleanParam(name: 'SKIP_ECR_CHECK', defaultValue: false, description: '跳过ECR Public镜像存在校验（应急勾选）')
        booleanParam(name: 'SKIP_ARGO_SYNC', defaultValue: false, description: '仅提交Git，不触发ArgoCD同步（调试用）')
    }
    environment {
        AWS_CREDS = credentials('aws-ecr-cred')
        GIT_USER = "rollback-ci"
        GIT_EMAIL = "rollback-ci@xxx.com"
        OPS_GIT_URL = 'git@gitlab.xx.com:gitops/ops-deploy.git'

        ARGO_APP = "${SERVICE_NAME}-${DEPLOY_ENV}"
        ARGOCD_SERVER   = 'argocd.xxx.com:80'
        GITOPS_SHA = 'HEAD'

        KUSTOMIZE_FILE = "./${BUSINESS}/${SERVICE_NAME}/overlays/${DEPLOY_ENV}/kustomization.yaml"
        ECR_PUBLIC_REPO_URI = "public.ecr.aws/xxx/${SERVICE_NAME}"
        GIT_COMMIT_MSG = "rollback ${params.SERVICE_NAME} to image tag ${ROLLBACK_TAG}"
    }
    stages {
        stage('0. 输入合法性校验') {
            steps {
                script { currentBuild.displayName = "#${BUILD_NUMBER} ${SERVICE_NAME}:ROLLBACK_TAG by ${env.BUILD_USER}" }

                sh '''
                if [ -z "${SERVICE_NAME}" ] || [ -z "${ROLLBACK_TAG}" ] || [ -z "${BUSINESS}" ] || [ -z "${DEPLOY_ENV}" ]; then
                    echo "❌ SERVICE_NAME / ROLLBACK_TAG / BUSINESS / DEPLOY_ENV 不能为空"
                    exit 1
                fi
                echo """✅ 参数校验通过
服务: ${SERVICE_NAME}
目标回滚TAG: ${ROLLBACK_TAG}
ECR Public仓库: ${ECR_PUBLIC_REPO_URI}
Kustomize文件: ${KUSTOMIZE_FILE}
                """
                '''
            }
        }
        stage('1. 镜像 Tag 存在校验') {
            when {
                expression { !params.SKIP_ECR_CHECK }
            }
            steps {
                withCredentials([string(credentialsId: 'aws-ecr-cred', variable: 'AWS_CREDS')]) {
                    sh '''
                    export AWS_ACCESS_KEY_ID=$(echo ${AWS_CREDS} | cut -d';' -f1)
                    export AWS_SECRET_ACCESS_KEY=$(echo ${AWS_CREDS} | cut -d';' -f2)
                    # ECR Public region固定 us-east-1
                    ECR_PUBLIC_REPO_NAME=${ECR_PUBLIC_REPO_URI#public.ecr.aws/}
                    ECR_PUBLIC_REPO_NAME=${ECR_PUBLIC_REPO_NAME#*/}
                    echo "检查 ECR Public 镜像 ${ECR_PUBLIC_REPO_URI}:${ROLLBACK_TAG}"
                    aws ecr-public describe-images \
                        --region us-east-1 \
                        --repository-name ${ECR_PUBLIC_REPO_NAME} \
                        --image-ids imageTag=${ROLLBACK_TAG}
                    if [ $? -eq 0 ]; then
                        echo "✅ ECR Public镜像tag ${ROLLBACK_TAG} 存在"
                    else
                        echo "❌ ECR Public镜像tag ${ROLLBACK_TAG} 不存在！终止回滚"
                        exit 1
                    fi
                    '''
                }
            }
        }
        stage('2. Checkout 部署仓库') {
            steps {
                git branch: 'main', credentialsId: 'git-argocd', url: OPS_GIT_URL

                sh '''
                if [ ! -f "${KUSTOMIZE_FILE}" ]; then
                    echo "❌ kustomization.yaml 文件 ${KUSTOMIZE_FILE} 不存在，请检查路径！"
                    exit 1
                fi
                '''
            }
        }
        stage('3. 更新 newTag') {
            steps {
                sh """
                # 获取当前配置里的 newTag
                CURRENT_TAG=\$(yq eval '.images[] | select(.name == strenv(ECR_PUBLIC_REPO_URI)) | .newTag' ${KUSTOMIZE_FILE})
                echo "当前kustomize内newTag: \${CURRENT_TAG}"
                echo "目标回滚TAG: ${ROLLBACK_TAG}"
                # 备份原文件
                cp ${KUSTOMIZE_FILE} ${KUSTOMIZE_FILE}.bak
                # yq v4 修改匹配 name 的 images 条目 newTag
                yq eval '.images[] |= (select(.name == strenv(ECR_PUBLIC_REPO_URI)) | .newTag = strenv(ROLLBACK_TAG))' -i ${KUSTOMIZE_FILE}
                echo "===== 修改前后对比 =====
旧 newTag: \${CURRENT_TAG}
新 newTag: \$(yq eval '.images[] | select(.name == strenv(ECR_PUBLIC_REPO_URI)) | .newTag' ${KUSTOMIZE_FILE})
========================"
                # yaml语法校验
                yq eval ${KUSTOMIZE_FILE} > /dev/null
                rm -f ${KUSTOMIZE_FILE}.bak
                """
            }
        }
        stage('4. 人工确认回滚操作') {
            steps {
                script {
                    def currentTag = sh(
                        script: """
                        yq eval '.images[] | select(.name == strenv(ECR_PUBLIC_REPO_URI)) | .newTag' ${KUSTOMIZE_FILE}
                        """,
                        returnStdout: true
                    ).trim() // 去掉末尾换行符

                    input message: """
⚠️ 回滚确认（Kustomize + ECR Public）
> 🔒 仅 jasper / abc / def 用户可以审批本次回滚！

服务：${SERVICE_NAME}
当前配置镜像TAG：${currentTag}
目标回滚镜像TAG：${ROLLBACK_TAG}
ECR Public仓库：${ECR_PUBLIC_REPO_URI}
Kustomize文件：${KUSTOMIZE_FILE}
ArgoCD App：${ARGO_APP}

确认提交Git并执行回滚吗？
""",
                    ok: "确认执行回滚",
                    submitter: 'jasper,bx,lemon'
                }
            }
        }
        stage('5. Git提交并推送ops-deploy') {
            steps {
                script {
                    withCredentials([sshUserPrivateKey(credentialsId: 'git-argocd', keyFileVariable: 'PK')]) {
                      sh """
                      ls -l
                      git status
                      git config user.name ${GIT_USER}
                      git config user.email ${GIT_EMAIL}
                      GIT_SSH_COMMAND="ssh -i $PK" git pull origin main
                      git add ${KUSTOMIZE_FILE}
                      git commit -m "${GIT_COMMIT_MSG}"
                      GIT_SSH_COMMAND="ssh -i $PK" git push origin HEAD:main
                      echo "✅ Git推送完成"
                      """
                    }
                }
            }
        }
        stage('6. ArgoCD 同步并等待健康') {
            steps {
                withCredentials([
                    string(
                        credentialsId: 'argocd-token',
                        variable: 'ARGOCD_AUTH_TOKEN'
                    )
                ]) {
                    sh '''
                        echo "$ARGOCD_AUTH_TOKEN"
                        argocd app sync "$ARGO_APP" \
                          --server "$ARGOCD_SERVER" \
                          --grpc-web \
                          --plaintext \
                          --auth-token "$ARGOCD_AUTH_TOKEN" \
                          --revision "$GITOPS_SHA" \
                          --timeout 300

                        argocd app wait "$ARGO_APP" \
                          --server "$ARGOCD_SERVER" \
                          --grpc-web \
                          --plaintext \
                          --auth-token "$ARGOCD_AUTH_TOKEN" \
                          --sync \
                          --health \
                          --operation \
                          --timeout 300

                        echo "✅ ArgoCD同步完成，应用健康"
                    '''
                }
            }
        }

    }
    post {
        success {
            echo "🎉 回滚任务执行成功！服务 ${SERVICE_NAME} 已回滚至镜像tag ${ROLLBACK_TAG}"
            script { notifyLark('SUCCESS') }
        }
        failure {
            echo "❌ 回滚任务失败，请检查日志！服务 ${SERVICE_NAME}"
            script { notifyLark('FAILURE') }
        }
        aborted {
            echo "⚠️ 回滚任务被取消或者非授权用户拒绝审批"
            script { notifyLark('ABORTED') }
        }
        always {
            cleanWs()
        }
    }
}

1.参数优化(无效果)





#export MAVEN_OPTS=-Xms4096m -Xmx4096m
export MAVEN_OPTS='-XX:-TieredCompilation -XX:TieredStopAtLevel=1'
mvn -T 2C -B -Dmaven.test.skip=true -P <xxx> clean package -f pom.xml






    

  • export MAVEN_OPTS='-XX:-TieredCompilation -XX:TieredStopAtLevel=1' 禁用默认分层编译
    • 

  • -B 以批处理(batch)模式运行
    • 

  • -T 2C 每一个CPU核使用2个线程，逻辑CPU核数. 模块化编译效果更佳 -pl module1,module2 -am
    • 

  • -Dmaven.test.skip = true 跳过单元测试。使用 -DskipTests 选项跳过测试执行，但仍会编译 test 文件夹
    • 

  • -Dmaven.compile.fork=true 指明多线程进行编译
    • 

  • clean 通常情况下，我们建议用户在构建时使用 clean 参数保证构建的正确性。clean 可以删除旧的构建产物，但其实我们大多数时间可能不需要这个参数，只有在某些情况下（比如，更改了类名，或者删除了一些类）才必须使用这个参数，所以，如果某次变更只是修改了一些方法，或者增加了一些类，那么就不需要强制执行 clean 了
    •
