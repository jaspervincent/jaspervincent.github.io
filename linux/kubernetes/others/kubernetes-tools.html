MetaLB

轻量级负载均衡(类型为 LoadBalancer 的服务)

https://metallb.universe.tf/

安装 

kubectl apply -f https://raw.githubusercontent.com/metallb/metallb/v0.14.8/config/manifests/metallb-native.yaml


#定义分配给负载均衡器服务的 IP
kubectl apply -f <<-EOF
apiVersion: metallb.io/v1beta1
kind: IPAddressPool
metadata:
  name: first-pool
  namespace: metallb-system
spec:
  addresses:
  - 192.168.10.0/24
EOF

#二层配置
kubectl apply -f <<-EOF
apiVersion: metallb.io/v1beta1
kind: L2Advertisement
metadata:
  name: example
  namespace: metallb-system
spec:
  ipAddressPools:
  - first-pool
EOF
