#-------------检查网络线路是否通畅------------------
function checkNetwork(){
    PkgServerIP=`echo ${PkgServer} |awk -F':' '{print $1}'`
    PkgServerPort=`echo ${PkgServer} |awk -F':' '{print $2}'`
    pkgNum=`echo -n "\n"|telnet ${PkgServerIP} ${PkgServerPort}|grep Connected|wc -l`
    if [ $pkgNum -ne 1 ]
    then
        echo "pkgNum is $pkgNum" >> ${LOG_DIR}
        exit 1
    fi
}

#5#Close the selinux and iptbles
function selinux_safety(){
if [ $(grep -i '=disabled' /etc/selinux/config|wc -l) -lt 1]
then
sed -i s#SELINUX=enforcing#SELINUX=disabled#g /etc/selinux/config
setenforce 0
getenforce
sleep 1
iptables –F
/etc/init.d/iptables save
fi
}

#6#Adjust the number of file descriptors
function limit_char(){
if [ $(grep '65535' /etc/security/limits.conf|wc -l) -lt 1 ]
then
echo '*               -       nofile          65535 ' >>/etc/security/limits.conf
fi
}

#7#Timing automatic cleaning/var/spool/clientmquene/junk files, prevent inodes filled nodes (c6.4 default no sendmail, therefore, can not configuration)
function remove_spam(){
if [ $(grep 'del_sys_file.sh' /var/spool/cron/root|wc -l) -lt 1 ]
then
mkdir /server/scripts -p &>/etc/dull
echo "find /var/spool/clientmquene/ -type f|xargs rm -f " >/server/scripts/del_sys_file.sh
echo "#del clientmqueue files by xu at 2014.10">>/var/spool/cron/root
echo "00 00 * * 0 /bin/sh /server/scripts/del_sys_file.sh >/dev/null 2>&1">>/var/spool/cron/root
fi
}

#8#service configuration

net.ipv4.tcp_tw_recycle = 1

net.ipv4.tcp_synack_retries = 1

===================================
net.nf_conntrack_max = 25000000
