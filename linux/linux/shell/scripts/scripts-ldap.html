ldap备份

备份目录 /apps/ops/ldap_backup/ldap

crontab 

# 备份ldap用户信息到git
0 1 * * * bash /apps/ops/ldap_backup/ldap/backup_ldap.sh

脚本 

cat <<\EOF> backup_ldap.sh
#!/bin/bash
LDAPBK=ldap-user-backup.ldif
BACKUPDIR=/apps/ops/ldap_backup/ldap

backuping() {
    echo "Backup Ldap Start...."
    /usr/sbin/slapcat -v -l ${BACKUPDIR}/${LDAPBK}
}

backuping
cd ${BACKUPDIR};git pull origin master;git add .;git commit -am "auto-backup-$(date +%Y-%m-%d)";git push origin master
EOF
