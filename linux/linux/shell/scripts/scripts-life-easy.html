调用函数

利用自定义函数做一些事情 

cat <<\EOF>> ~/bin/@f
setup_00_print_hello() {
  echo hello
}

main() {
  declare -F |
    awk '{print $3;}' |
    grep ^setup_ |
    while read -r fn; do
      local _thing=
      _thing=${fn//setup_/}
      _thing=${_thing//_/ }
      printf "👾 %s ...\\n" "${_thing}"
      "${fn}"
    done
}

main
EOF

chmod +x @f

export PATH=~/bin/:$PATH

]# @f 
👾 00 print hello ...
hello

setup_00_screens() {
  local cwd=
  cwd="$(cd "$(dirname "${BASH_SOURCE[0]}")" &> /dev/null && pwd)"

setup_05_network_dns() {

setup_20_stop_vms() {
  virsh list | grep running | awk '{print $2}' |
    while read -r vm; do
      virsh shutdown "${vm}" &> /dev/null &
    done
}

setup_20_daily() {
  @daily
}

setup_01_network() {
  local _interface=wlan0
  iwctl station wlan0 disconnect
  iwctl station wlan0 connect Stibo
  command dhclient &> /dev/null && sudo dhclient -v "${_interface}"

  local _ip _domain
  _ip=$(ip address show "${_interface}" |
          sed -nr 's#.* inet ([^/]*)/[0-9]+ .*#\1#p')
  _domain=$(iwctl station "${_interface}" show |
    sed -n 's/Connected network//p' |
    xargs)
  printf "🚀 Connected with IP %s in domain %s ...\\n" \
         "${_ip}" \
         "${_domain}"

  test -r /etc/resolv.conf.work &&
    sudo cp /etc/resolv.conf.work /etc/resolv.conf
}


main "$@"

#! /usr/bin/env bash

log="$HOME/.$(basename "$0")".last.log

main() {
  echo >"${log}"

  declare -F | sed -rn 's#.* (cmd_.*)#\1#p' | sort -n |
    while read -r fn; do
      local _thing=
      _thing=${fn//cmd_/}
      _thing=${_thing//_/ }
      printf "🐉 %s ...\\n" "${_thing}"
      "${fn}"
    done
}

cmd_00_ensure_sudo_is_ready_to_use() {
  sudo true
}

cmd_02_private() {
  test -x ~/bin/@daily.private && ~/bin/@daily.private
}

disabled_03_update_emacs_src() {
  test -d /usr/local/src/emacs/.git && {
    git -C /usr/local/src/emacs pull --quiet
    git -C /usr/local/src/emacs log --oneline -n 1 --pretty='format:%ad %s' |
      sed -rn 's#(.*)#\t\1\n#p'
  }
}

cmd_80_update_os() {
  test -r /etc/pacman.conf && {
    yes | sudo pacman -Syu &>> "${log}"
  }
}

cmd_80_update_os_aur() {
  test -r /etc/paru.conf && {
    yes | paru --sudoloop -Syu &>> "${log}"
  }
}

_cmd_00_setup_ssh_agent() {
  # First, check for existing agent
  _agent=$(find /tmp/ssh-* -name "agent*" | tail -n 1)
  if [ -z "${_agent}" ]; then
    ssh-agent &
  fi

  # Now, there *should* be an agent, for sure.
  _agent=$(find /tmp/ssh-* -name "agent*" | tail -n 1)

  # Set symlink so that ~/.emacs can refer to it
  ln -sf "${_agent}" /tmp/ssh-agent
  export SSH_AUTH_SOCK=/tmp/ssh-agent
  test -x /usr/lib/ssh/ssh-askpass || {
    printf "Need to install: %s, or else, ssh-add will not be able to add keys with password. \\n" \
           /usr/lib/ssh/ssh-askpass |
      xargs |
      fold
  }

  ssh-add -l | grep -q .ssh/id_ || ssh-add
}

cmd_85_backup_shell_history() {
  local _history=$HOME/.zsh_history
  cp "${_history}" "${_history}.old"
}

cmd_90_check_if_restart_is_desired() {
  local running_version=
  running_version=$(uname -r)

  test -d /lib/modules/"${running_version}" || {
    printf "🐉 Kernel upgraded rebooting system\\n"
    echo sudo reboot
  }
}

main "$@"
